import styled, { css } from 'styled-components';

const Button = styled.button<{ $primary?: boolean }>`
    background: transparent;
    color: black;
    ${(props) =>
        props.$primary &&
        css`
            background: red;
            color: white;
        `
    }
`;

export default function App(){
    return ______Click Me</Button>;
}
